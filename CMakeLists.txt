cmake_minimum_required(VERSION 3.20)
project(EfficientEdgeAgent VERSION 0.1.0 LANGUAGES C CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64)$")
  message(FATAL_ERROR "Only Linux x86-64 is supported")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_library(edge_sqlite STATIC third_party/sqlite3.c)
target_include_directories(edge_sqlite PUBLIC third_party)
target_compile_definitions(edge_sqlite PRIVATE SQLITE_THREADSAFE=1 SQLITE_OMIT_LOAD_EXTENSION SQLITE_DQS=0)
target_link_libraries(edge_sqlite PUBLIC Threads::Threads ${CMAKE_DL_LIBS} m)
add_library(edge_core src/common.cpp src/tools.cpp src/inference.cpp src/agent.cpp src/config.cpp)
target_include_directories(edge_core PUBLIC include)
target_include_directories(edge_core SYSTEM PUBLIC third_party)
target_compile_options(edge_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
target_compile_definitions(edge_core PUBLIC EDGE_VERSION="${PROJECT_VERSION}")
target_link_libraries(edge_core PUBLIC edge_sqlite Threads::Threads)
add_executable(edge-agent src/main.cpp)
target_link_libraries(edge-agent PRIVATE edge_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(edge-tests tests/test_main.cpp)
  target_link_libraries(edge-tests PRIVATE edge_core)
  add_test(NAME unit COMMAND edge-tests ${CMAKE_CURRENT_SOURCE_DIR})
  set_tests_properties(unit PROPERTIES LABELS "model-independent" TIMEOUT 30)
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_FOUND)
    add_test(NAME http-contract COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/http_contract.py $<TARGET_FILE:edge-agent> ${CMAKE_CURRENT_SOURCE_DIR})
    set_tests_properties(http-contract PROPERTIES LABELS "model-independent" TIMEOUT 30)
  endif()
  option(EDGE_REAL_MODEL_TEST "Enable live Qwen3 server integration test" OFF)
  if(EDGE_REAL_MODEL_TEST AND Python3_FOUND)
    add_test(NAME qwen3-live COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/scripts/live_smoke.py --binary $<TARGET_FILE:edge-agent>)
    set_tests_properties(qwen3-live PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR} LABELS "model-dependent" TIMEOUT 240)
  endif()
endif()
